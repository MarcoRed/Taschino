function detectLanguage(text) {
  const words = (text.toLowerCase().match(/[a-zàèéìòù']+/g) || []);
  if (words.length < 8) return null; // testo troppo corto per un giudizio affidabile
  let itScore = 0, enScore = 0;
  words.forEach(w => {
    if (IT_MARKERS.includes(w)) itScore++;
    else if (EN_MARKERS.includes(w)) enScore++;
  });
  if (itScore === 0 && enScore === 0) return null;
  return enScore > itScore ? 'en' : 'it';
}

function applyDetectedLanguage() {
  if (!hasTTS || voices.length === 0) return;
  const text = textInput.value;
  const lang = detectLanguage(text);
  if (!lang) { langNote.textContent = ''; return; }
  const matchIndex = voices.findIndex(v => v.lang.toLowerCase().startsWith(lang));
  if (matchIndex === -1) {
    langNote.textContent = 'Lingua rilevata: ' + (lang === 'en' ? 'Inglese' : 'Italiano') + ' — nessuna voce disponibile per questa lingua, uso quella selezionata.';
    return;
  }
  voiceSelect.value = String(matchIndex);
  langNote.textContent = 'Lingua rilevata: ' + (lang === 'en' ? 'Inglese' : 'Italiano') + ' — voce impostata su ' + voices[matchIndex].name + '.';
}

function stripMarkdown(text) {
  return text
    .replace(/```[\s\S]*?```/g, '')           // blocchi di codice
    .replace(/`([^`]+)`/g, '$1')               // codice inline
    .replace(/!\[([^\]]*)\]\([^)]*\)/g, '$1')  // immagini -> testo alt
    .replace(/\[([^\]]+)\]\([^)]*\)/g, '$1')   // link -> testo del link
    .replace(/^#{1,6}\s+/gm, '')               // titoli
    .replace(/(\*\*|__)(.*?)\1/g, '$2')        // grassetto
    .replace(/(\*|_)(.*?)\1/g, '$2')           // corsivo
    .replace(/^>\s?/gm, '')                    // citazioni
    .replace(/^\s*[-*+]\s+/gm, '')             // elenchi puntati
    .replace(/^\s*\d+\.\s+/gm, '')             // elenchi numerati
    .replace(/^\s*\|?[\s:|-]{3,}\|?\s*$/gm, '') // righe di separazione tabelle (|---|---|---|)
    .replace(/^-{3,}\s*$/gm, '')               // linee orizzontali
    .replace(/\|/g, ' ')                       // separatori tabelle
    .replace(/\n{3,}/g, '\n\n')                // righe vuote multiple
    .trim();
}

function splitIntoSentences(text) {
  // Divide per punto/esclamativo/interrogativo seguiti da spazio o fine riga, mantenendo il punteggiatura
  const parts = text.match(/[^.!?\n]+[.!?\n]*/g) || [text];
  return parts.map(s => s.trim()).filter(s => s.length > 0);
}

function prepareReadingBox(text) {
  readingBox.innerHTML = '';
  const words = text.split(/(\s+)/);
  wordSpans = [];
  words.forEach(w => {
    if (w.trim() === '') {
      readingBox.appendChild(document.createTextNode(w));
    } else {
      const span = document.createElement('span');
      span.textContent = w;
      readingBox.appendChild(span);
      wordSpans.push(span);
    }
  });
  readingBox.classList.add('visible');
}

function clearHighlight() {
  wordSpans.forEach(s => { s.style.background = ''; s.style.color = ''; });
}

function highlightWordAt(globalWordIndex) {
  clearHighlight();
  const span = wordSpans[globalWordIndex];
  if (span) {
    span.style.background = '#f2e2dc';
    span.style.color = '#b5563c';
    span.style.borderRadius = '4px';
    span.scrollIntoView({ block: 'center', behavior: 'smooth' });
  }
}

// Precalcola, per ogni frase, l'indice della prima parola globale
let sentenceWordOffsets = [];

function buildSentenceOffsets() {
  sentenceWordOffsets = [];
  let count = 0;
  sentenceQueue.forEach(s => {
    sentenceWordOffsets.push(count);
    count += s.split(/\s+/).filter(w => w.length > 0).length;
  });
}

function speakSentence(index) {
  if (!isReading) return;
  if (index >= sentenceQueue.length) {
    ttsStatus.textContent = 'Lettura terminata.';
    clearHighlight();
    isReading = false;
    releaseWakeLock();
    hideDrivingMode();
    return;
  }
  currentSentenceIndex = index;
  speechGeneration++;
  const myGeneration = speechGeneration;

  const sentenceText = sentenceQueue[index];
  const utterance = new SpeechSynthesisUtterance(sentenceText);
  const selectedVoice = voices[voiceSelect.value];
  if (selectedVoice) utterance.voice = selectedVoice;
  utterance.rate = parseFloat(rateRange.value);
  utterance.lang = selectedVoice ? selectedVoice.lang : 'it-IT';

  const baseWordOffset = sentenceWordOffsets[index];
  currentGlobalWordIndex = baseWordOffset; // granularità minima: inizio frase (alcuni motori Android non emettono onboundary)
  let advanced = false; // protezione: alcuni motori vocali Android scatenano onend due volte per la stessa frase

  function advanceOnce() {
    if (advanced) return;
    advanced = true;
    if (isReading && myGeneration === speechGeneration) speakSentence(index + 1);
  }

  utterance.onboundary = (event) => {
    if (myGeneration !== speechGeneration) return; // frase superata da un salto: ignora
    if (event.name === 'word') {
      const before = sentenceText.slice(0, event.charIndex);
      const wordsBefore = before.split(/\s+/).filter(w => w.length > 0).length;
      currentGlobalWordIndex = baseWordOffset + wordsBefore;
      highlightWordAt(currentGlobalWordIndex);
    }
  };

  utterance.onend = () => {
    if (myGeneration !== speechGeneration) return; // già superata da un salto
    advanceOnce();
  };

  utterance.onerror = (event) => {
    if (myGeneration !== speechGeneration) return; // già superata da un salto
    if (advanced) return; // evita di mostrare/ripetere due volte lo stesso errore
    ttsStatus.textContent = 'Errore alla frase ' + (index + 1) + '/' + sentenceQueue.length + ' (' + (event.error || 'sconosciuto') + '). Provo a proseguire...';
    advanceOnce();
  };

  ttsStatus.textContent = 'Lettura in corso... (frase ' + (index + 1) + '/' + sentenceQueue.length + ')';
  drivingProgress.textContent = 'Frase ' + (index + 1) + ' di ' + sentenceQueue.length;
  updateDrivePlayPauseIcon();
  speechSynthesis.speak(utterance);
}

function showDrivingMode() {

}

function estimateWordsPerSecond() {
  const rate = parseFloat(rateRange.value) || 1;
  const baseWordsPerMinute = 150; // stima di lettura naturale a velocità 1.0x, regola se non torna con i tuoi tempi reali
  return (baseWordsPerMinute * rate) / 60;
}

function findSentenceIndexForWord(targetWordIndex) {
  let idx = 0;
  for (let i = 0; i < sentenceWordOffsets.length; i++) {
    if (sentenceWordOffsets[i] <= targetWordIndex) idx = i;
    else break;
  }
  return idx;
}

function jumpSeconds(deltaSeconds) {
  if (!isReading || sentenceQueue.length === 0) return;
  const wasPaused = speechSynthesis.paused;
  const wordsPerSecond = estimateWordsPerSecond();
  const deltaWords = Math.round(deltaSeconds * wordsPerSecond);
  const totalWords = wordSpans.length;
  let targetWord = currentGlobalWordIndex + deltaWords;
  targetWord = Math.max(0, Math.min(targetWord, Math.max(totalWords - 1, 0)));
  const targetSentenceIndex = findSentenceIndexForWord(targetWord);
  speechSynthesis.cancel();
  speakSentence(targetSentenceIndex);
  if (wasPaused) speechSynthesis.pause();
}

#!/usr/bin/env bash
#
# merge_project.sh — unisce ricorsivamente tutti i file di testo di un progetto
# in un unico file, con intestazioni che indicano il percorso relativo di ciascun
# file. Pensato per creare un file da caricare nella "project knowledge" di Claude.
#
# USO:
#   ./merge_project.sh [cartella_progetto] [file_output]
#
# ESEMPI:
#   ./merge_project.sh                          # usa la cartella corrente, output: project_merged.txt
#   ./merge_project.sh ~/taschino                # cartella specifica, output: project_merged.txt
#   ./merge_project.sh ~/taschino merged.md      # cartella e nome output specifici

set -euo pipefail

SRC_DIR="${1:-.}"
OUT_FILE="${2:-project_merged.txt}"

# Cartelle/pattern da escludere sempre (aggiungi qui se ti serve)
EXCLUDE_DIRS=(".git" "node_modules" "dist" "build" ".venv" "__pycache__" ".idea" ".vscode")

# Costruisce l'espressione find per escludere le cartelle sopra
PRUNE_EXPR=()
for d in "${EXCLUDE_DIRS[@]}"; do
  PRUNE_EXPR+=(-path "*/$d" -o)
done
# rimuove l'ultimo -o
unset 'PRUNE_EXPR[${#PRUNE_EXPR[@]}-1]'

# Svuota/crea il file di output
: > "$OUT_FILE"

# Intestazione generale col riepilogo dell'albero delle directory (utile come mappa)
{
  echo "# Struttura del progetto: $SRC_DIR"
  echo "# Generato il: $(date '+%Y-%m-%d %H:%M:%S')"
  echo
  echo "## Albero file (esclusi: ${EXCLUDE_DIRS[*]})"
  echo '```'
} >> "$OUT_FILE"

find "$SRC_DIR" \( "${PRUNE_EXPR[@]}" \) -prune -o -type f -print \
  | sed "s|^$SRC_DIR/||" | sort >> "$OUT_FILE"

echo '```' >> "$OUT_FILE"
echo >> "$OUT_FILE"
echo "---" >> "$OUT_FILE"
echo >> "$OUT_FILE"

# Trova tutti i file, escludendo le cartelle indicate, ordinati per percorso
FILES=$(find "$SRC_DIR" \( "${PRUNE_EXPR[@]}" \) -prune -o -type f -print | sort)

while IFS= read -r file; do
  # Salta il file di output stesso se si trova dentro SRC_DIR
  [ "$(realpath "$file")" = "$(realpath "$OUT_FILE")" ] && continue

  # Salta i file binari (immagini, font, ecc.) — include solo testo
  if ! file -b --mime "$file" | grep -qE 'text|json|xml|javascript|x-empty'; then
    continue
  fi

  rel_path="${file#$SRC_DIR/}"
  lines=$(wc -l < "$file" | tr -d ' ')
  size=$(wc -c < "$file" | tr -d ' ')

  {
    echo "=== FILE: $rel_path ==="
    echo "# righe: $lines — dimensione: ${size} byte"
    echo
    cat "$file"
    echo
    echo "=== FINE FILE: $rel_path ==="
    echo
  } >> "$OUT_FILE"
done <<< "$FILES"

echo "Fatto. Output scritto in: $OUT_FILE"
echo "Righe totali: $(wc -l < "$OUT_FILE")"

# Taschino

Piccola collezione di mini-app HTML personali, pubblicate come sito statico via GitLab Pages.

## Struttura

```
taschino/
├── index.html              # home page con l'elenco dei tool
├── .gitlab-ci.yml           # configurazione per GitLab Pages
└── lettore-vocale/
    └── index.html           # tool: lettore vocale (testo o link → voce)
```

## Come pubblicare

1. Crea un nuovo repository vuoto su GitLab, es. `taschino`.
2. Nella cartella di questo progetto:

```bash
git init
git remote add origin git@gitlab.com:<tuo-utente>/taschino.git
git add .
git commit -m "Primo tool: lettore vocale"
git branch -M main
git push -u origin main
```

3. Vai su **Settings → Pages** nel progetto GitLab: dopo la prima pipeline (Build → Pipelines), la Pages sarà attiva automaticamente grazie al file `.gitlab-ci.yml` incluso.
4. L'URL sarà del tipo:

```
https://<tuo-utente>.gitlab.io/taschino/
```

## Aggiungere un nuovo tool

1. Crea una nuova cartella, es. `nuovo-tool/`, con dentro un `index.html`.
2. Aggiungi una card nella home page (`index.html` principale) che punta a `nuovo-tool/`.
3. Per renderlo installabile come app (icona propria, schermo intero, no barra Chrome):
   - Crea `nuovo-tool/manifest.json` (copia quello di `lettore-vocale/` e cambia nome/descrizione)
   - Aggiungi due icone `icon-192.png` e `icon-512.png` nella cartella
   - Nell'`<head>` dell'`index.html` del tool, aggiungi gli stessi tag `<link rel="manifest">`, `<meta name="theme-color">`, `<link rel="apple-touch-icon">` presenti in `lettore-vocale/index.html`
4. Commit e push: la pipeline ripubblica tutto automaticamente.

## Installare un tool come app sul telefono

Apri l'URL del tool specifico (es. `.../lettore-vocale/`) da Chrome → menu (⋮) → **"Installa app"** (o "Aggiungi a schermata Home" se non compare l'opzione diretta). Su iPhone/Safari: pulsante Condividi → **"Aggiungi a Home"**. Con il manifest configurato, l'icona sarà quella personalizzata e l'app si aprirà a schermo intero, senza barra degli indirizzi.
